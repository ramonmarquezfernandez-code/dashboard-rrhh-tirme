# Ranking HE combo, ejercicio 2025 · perfil empleado · sentencia 1

SELECT zparte.`PERNR` AS pernr, userpayroll.`NAME` AS nombre, userpayroll.`SURNAME` AS apellidos, coalesce(sum(zparte.`DLCOP` + zparte.`DFCOP` + zparte.`NLCOP` + zparte.`NFCOP`), 0) AS combo_programadas, coalesce(sum(zparte.`DL` + zparte.`DF` + zparte.`NL` + zparte.`NF` + zparte.`DLC` + zparte.`DFC` + zparte.`NLC` + zparte.`NFC` + zparte.`DLB` + zparte.`DFB` + zparte.`NLB` + zparte.`NFB` + zparte.`DLBN` + zparte.`DFBN` + zparte.`NLBN` + zparte.`NFBN` + zparte.`DLCO` + zparte.`DFCO` + zparte.`NLCO` + zparte.`NFCO` + zparte.`DLF` + zparte.`DFF` + zparte.`NLF` + zparte.`NFF` + zparte.`DLCOP` + zparte.`DFCOP` + zparte.`NLCOP` + zparte.`NFCOP`), 0) AS total_he, coalesce(sum(zparte.`DL` + zparte.`DF` + zparte.`NL` + zparte.`NF`), 0) AS he_compensables, coalesce(sum(zparte.`DL` * 1.6e0 + zparte.`DF` * 2.0e0 + zparte.`NL` * 2.0e0 + zparte.`NF` * 2.5e0), 0) AS he_compensables_convertidas 
FROM zparte LEFT OUTER JOIN userpayroll ON zparte.`PERNR` = userpayroll.`NUMPER` 
WHERE zparte.`PERNR` = '00000012' AND zparte.`EJERC` = '2025' AND zparte.`MES` IN ('01', '02', '03', '04', '05', '06', '07', '08', '09', '10', '11', '12') GROUP BY zparte.`PERNR`, userpayroll.`NAME`, userpayroll.`SURNAME` ORDER BY sum(zparte.`DL` + zparte.`DF` + zparte.`NL` + zparte.`NF` + zparte.`DLC` + zparte.`DFC` + zparte.`NLC` + zparte.`NFC` + zparte.`DLB` + zparte.`DFB` + zparte.`NLB` + zparte.`NFB` + zparte.`DLBN` + zparte.`DFBN` + zparte.`NLBN` + zparte.`NFBN` + zparte.`DLCO` + zparte.`DFCO` + zparte.`NLCO` + zparte.`NFCO` + zparte.`DLF` + zparte.`DFF` + zparte.`NLF` + zparte.`NFF` + zparte.`DLCOP` + zparte.`DFCOP` + zparte.`NLCOP` + zparte.`NFCOP`) DESC, zparte.`PERNR` ASC
