# SP de los últimos 12 y 24 meses · perfil empleado · sentencia 1

SELECT userpayroll.`NAME` AS nombre, userpayroll.`SURNAME` AS apellidos, zparte.`PERNR` AS id, zparte.`DPTO` AS departamento, count(*) AS sp 
FROM zparte LEFT OUTER JOIN userpayroll ON zparte.`PERNR` = userpayroll.`NUMPER` 
WHERE zparte.`PERNR` = '00000012' AND zparte.`PADAT` BETWEEN '2025-10-03' AND '2026-10-03' AND zparte.`SP` IN ('t', 'X', '1', 'S') GROUP BY zparte.`PERNR`, userpayroll.`NAME`, userpayroll.`SURNAME`, zparte.`DPTO` ORDER BY count(*) DESC, zparte.`PERNR` ASC
